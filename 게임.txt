        function updateStep() {
            if (vnEngineState.currentStep >= storySteps.length) return;
            const step = storySteps[vnEngineState.currentStep];

            // Clean previous dialogue animations & states
            const dbox = document.getElementById('vn-dialogue-box');
            if (dbox) dbox.classList.add('hidden');

            // Handle step-specific BGM changes
            if (vnEngineState.currentStep === 12) {
                // The exact step customer demands a refill: Transition to Anger BGM
                soundManager.play('anger');
            } else if (vnEngineState.currentStep < 4 || (vnEngineState.currentStep >= 5 && vnEngineState.currentStep < 11)) {
                // Before Game 1 and between Game 1 & Game 2: Transition to Intro BGM
                soundManager.play('intro');
            }

            // Screen shake configuration
            if (step.background && step.background.includes('store_shake')) {
                const viewport = document.getElementById('vn-viewport');
                const vignette = document.getElementById('vignette-overlay');
                
                if (viewport) viewport.classList.add('shake-active');
                
                let opacity = 0.35;
                if (step.background === 'store_shake_2') opacity = 0.75;
                if (vignette) vignette.style.opacity = opacity;

                // Safe shake timeout: Exactly 1-second limit
                if (vnEngineState.shakeTimeout) clearTimeout(vnEngineState.shakeTimeout);
                vnEngineState.shakeTimeout = setTimeout(() => {
                    if (viewport) viewport.classList.remove('shake-active');
                    if (vignette) vignette.style.opacity = '0';
                }, 1000);
            } else {
                const viewport = document.getElementById('vn-viewport');
                const vignette = document.getElementById('vignette-overlay');
                if (viewport) viewport.classList.remove('shake-active');
                if (vignette) vignette.style.opacity = '0';
            }

            // Check non-dialogue stages
            if (step.type) {
                if (step.type === "game_1_intro") {
                    loadGame1Intro();
                    return;
                } else if (step.type === "time_lapse") {
                    loadTimeLapse();
                    return;
                } else if (step.type === "game_2_intro") {
                    loadGame2Intro();
                    return;
                } else if (step.type === "show_verdict_platform") {
                    soundManager.stopAll();
                    router.navigate('judgment', { stageId: 'vote' });
                    return;
                }
            }

            // Render backgrounds
            renderStoryBackground(step.background);

            // Render Actors
            renderStoryActors(step.characters);

            // Mapping state emotions
            const customerImg = document.getElementById('img-customer');
            const clerkImg = document.getElementById('img-clerk');

            if (customerImg) {
                if ([14, 16, 18, 20, 21, 23].includes(vnEngineState.currentStep)) {

        function renderStoryBackground(type) {
            const bgContainer = document.getElementById('vn-bg-illustration');
            if (!bgContainer) return;

            let opacity = "opacity-100";
            let overlayFilter = "";

            if (type === "store_dim") {
                overlayFilter = "brightness-50";
            } else if (type === "black") {
                opacity = "opacity-0";
            }

            const sceneEl = document.getElementById('vn-scene');
            sceneEl.className = `absolute inset-0 z-10 transition-colors duration-500 flex items-end justify-center px-4 pb-[150px] ${type === 'black' ? 'bg-black' : 'bg-slate-900'}`;

            if (type === "black") {
                bgContainer.innerHTML = "";
            } else {
                bgContainer.innerHTML = `
                    <div class="w-full h-full relative transition-all duration-500 ${opacity}">
                        <img src="https://cdn.phototourl.com/free/2026-05-21-95ca7e63-ffac-4e02-87f1-0a609be64c0e.png" 
                             alt="매장 배경" 
                             class="w-full h-full object-cover transition-all duration-500 ${overlayFilter}" 
                             onerror="this.style.display='none'" />
                        ${type === "store_dim" ? '<div class="absolute inset-0 bg-black/40"></div>' : ''}
                    </div>
                `;
            }
        }

        function renderStoryActors(actors) {
            const list = ['customer', 'boss', 'clerk'];
            list.forEach(a => {
                const el = document.getElementById(`char-${a}-container`);
                if (!el) return;

                if (actors.includes(a)) {
                    if (el.classList.contains('hidden')) {
                        el.classList.remove('hidden', 'character-pop-exit');
                        el.classList.add('character-pop-enter');
                    }
                } else {
                    if (!el.classList.contains('hidden') && !el.classList.contains('character-pop-exit')) {
                        el.classList.remove('character-pop-enter');
                        el.classList.add('character-pop-exit');
                        setTimeout(() => {
                            el.classList.add('hidden');
                        }, 400);
                    }
                }
            });
        }

        function typewriterEffect(fullText) {
            clearInterval(vnEngineState.textInterval);
            vnEngineState.isTyping = true;
            
            const textElem = document.getElementById('vn-text-content');
            textElem.textContent = "";
            let idx = 0;

            vnEngineState.textInterval = setInterval(() => {
                if (idx < fullText.length) {
                    textElem.textContent += fullText[idx];
                    idx++;
                } else {
                    clearInterval(vnEngineState.textInterval);
                    vnEngineState.isTyping = false;
                }
            }, 35);
        }

        function handleSceneClick() {
            // Unmute safety guard
            if (soundManager.currentBgm && soundManager.currentBgm.paused) {
                soundManager.currentBgm.play().catch(e => {});
            }

            if (vnEngineState.isTyping) {
                // Skip to show full text instantly
                clearInterval(vnEngineState.textInterval);
                const step = storySteps[vnEngineState.currentStep];
                document.getElementById('vn-text-content').textContent = step.text;
                vnEngineState.isTyping = false;
            } else {
                // Proceed next step
                vnEngineState.currentStep++;
                updateStep();
            }
        }

        function initColaPhysicsEngine() {
            const canvas = document.getElementById('cola-canvas');
            if (!canvas) return;
            const ctx = canvas.getContext('2d');

            vnEngineState.game1Active = true;
            vnEngineState.colaWaterLevel = 0;
            vnEngineState.colaStreamY = 0;
            vnEngineState.isPressingCola = false;
            vnEngineState.colaSettled = false;
            vnEngineState.bubbles = [];

            const startPress = (e) => {
                if (e.cancelable) e.preventDefault();
                if (vnEngineState.colaSettled) return;
                vnEngineState.isPressingCola = true;
            };

            const endPress = (e) => {
                vnEngineState.isPressingCola = false;
            };

            canvas.addEventListener('mousedown', startPress);
            canvas.addEventListener('touchstart', startPress, { passive: false });

            window.addEventListener('mouseup', endPress);
            window.addEventListener('touchend', endPress);

            function tick() {
                if (!vnEngineState.game1Active) {
                    window.removeEventListener('mouseup', endPress);
                    window.removeEventListener('touchend', endPress);
                    return;
                }

                const cupX = 64;
                const cupY = 100;
                const cupW = 128;
                const cupH = 180;
                const surfaceY = (cupY + cupH) - (cupH * (vnEngineState.colaWaterLevel / 100));

                if (vnEngineState.isPressingCola) {
                    if (vnEngineState.colaStreamY < surfaceY) {
                        vnEngineState.colaStreamY += 20; 
                        if (vnEngineState.colaStreamY > surfaceY) {
                            vnEngineState.colaStreamY = surfaceY;
                        }
                    } else {
                        vnEngineState.colaStreamY = surfaceY; 
                        vnEngineState.colaWaterLevel += 0.8; 
                    }
                } else {
                    if (vnEngineState.colaStreamY > 0) {
                        vnEngineState.colaStreamY += 24; 
                        if (vnEngineState.colaStreamY >= surfaceY) {
                            vnEngineState.colaStreamY = 0; 
                        }
                    }
                }

                if (vnEngineState.colaWaterLevel > 100) {
                    vnEngineState.colaWaterLevel = 100;
                }

                if (vnEngineState.colaWaterLevel > 0 && Math.random() < 0.35) {
                    vnEngineState.bubbles.push({
                        x: cupX + 20 + Math.random() * (cupW - 40),
                        y: (cupY + cupH) - Math.random() * (cupH * (vnEngineState.colaWaterLevel / 100)),
                        r: 1 + Math.random() * 2.5,
                        vy: 0.6 + Math.random() * 1.5
                    });
                }

                vnEngineState.bubbles.forEach((b, idx) => {
                    b.y -= b.vy;
                    const bubbleSurfaceY = (cupY + cupH) - (cupH * (vnEngineState.colaWaterLevel / 100));
                    if (b.y < bubbleSurfaceY - 5) {
                        vnEngineState.bubbles.splice(idx, 1);
                    }
                });

                // Clear with absolute transparency to see the gorgeous background image underneath
                ctx.clearRect(0, 0, canvas.width, canvas.height);

                ctx.fillStyle = "rgba(203, 213, 225, 0.8)";
                ctx.fillRect(118, 0, 20, 30);
                ctx.fillStyle = "#475569";
                ctx.fillRect(123, 25, 10, 8);

                ctx.fillStyle = "rgba(255, 255, 255, 0.4)";
                ctx.beginPath();
                ctx.moveTo(cupX, cupY);
                ctx.lineTo(cupX + 15, cupY + cupH);
                ctx.lineTo(cupX + cupW - 15, cupY + cupH);
                ctx.lineTo(cupX + cupW, cupY);
                ctx.closePath();
                ctx.fill();

                const targetFillY = (cupY + cupH) - (cupH * 0.8);
                ctx.strokeStyle = "#f43f5e";
                ctx.lineWidth = 3;
                ctx.setLineDash([4, 4]);
                ctx.beginPath();
                ctx.moveTo(cupX + 5, targetFillY);
                ctx.lineTo(cupX + cupW - 5, targetFillY);
                ctx.stroke();
                ctx.setLineDash([]);

                if (vnEngineState.colaWaterLevel > 0) {
                    const fillH = cupH * (vnEngineState.colaWaterLevel / 100);
                    const surfaceYPos = (cupY + cupH) - fillH;

                    const time = Date.now() * 0.006;
                    const waveAmp = vnEngineState.isPressingCola ? 3.5 : 1.2;

                    const leftX = cupX + 15 - 15 * (vnEngineState.colaWaterLevel / 100);
                    const rightX = cupX + cupW - 15 + 15 * (vnEngineState.colaWaterLevel / 100);

                    ctx.fillStyle = "rgba(59, 30, 16, 0.9)"; 
                    ctx.beginPath();
                    ctx.moveTo(cupX + 15, cupY + cupH);
                    ctx.lineTo(leftX, surfaceYPos);
                    
                    for (let wx = leftX; wx <= rightX; wx += 10) {
                        const wy = surfaceYPos + Math.sin((wx * 0.05) + time) * waveAmp;
                        ctx.lineTo(wx, wy);
                    }

                    ctx.lineTo(rightX, surfaceYPos);
                    ctx.lineTo(cupX + cupW - 15, cupY + cupH);
                    ctx.closePath();
                    ctx.fill();

                    ctx.fillStyle = "rgba(253, 224, 71, 0.75)"; 
                    vnEngineState.bubbles.forEach(b => {
                        ctx.beginPath();
                        ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2);
                        ctx.fill();
                    });
                }

                if (vnEngineState.colaStreamY > 0) {
                    const streamW = 6 + Math.sin(Date.now() * 0.05) * 1.5;
                    ctx.fillStyle = "#451a03";
                    ctx.fillRect(128 - streamW/2, 30, streamW, vnEngineState.colaStreamY - 30);

                    ctx.beginPath();
                    ctx.arc(128, vnEngineState.colaStreamY, 5 + Math.random()*5, 0, Math.PI*2);
                    ctx.fill();
                }

                ctx.strokeStyle = "rgba(71, 85, 105, 0.85)";
                ctx.lineWidth = 3.5;
                ctx.beginPath();
                ctx.moveTo(cupX, cupY);
                ctx.lineTo(cupX + 15, cupY + cupH);
                ctx.lineTo(cupX + cupW - 15, cupY + cupH);
                ctx.lineTo(cupX + cupW, cupY);
                ctx.stroke();

                if (!vnEngineState.isPressingCola && vnEngineState.colaWaterLevel > 0 && vnEngineState.colaStreamY === 0) {
                    if (!vnEngineState.colaSettled) {
                        vnEngineState.colaSettled = true;

            }

        function handleStainClick(e, spotId) {
            const spot = vnEngineState.stains.find(s => s.id === spotId);
            if (!spot || spot.clicks <= 0) return;

            spot.clicks--;
            
            const div = document.getElementById(`stain-spot-${spotId}`);
            if (div) {
                const scale = 0.4 + (spot.clicks / 15) * 0.6;
                div.style.transform = `translate(-50%, -50%) scale(${scale}) rotate(${Math.random() * 30 - 15}deg)`;
                
                const countLbl = document.getElementById(`stain-click-lbl-${spotId}`);
                if (countLbl) countLbl.textContent = spot.clicks;
            }

            spawnStarBurstEffect(e, div);

            if (spot.clicks === 0) {
                triggerMassiveCleanBurst(div);
                div.remove();
                
                vnEngineState.totalCleanCount++;

                if (vnEngineState.totalCleanCount === 3) {
                    clearInterval(vnEngineState.game2Interval);
                    evaluateGame2Success();
                }
            }
        }

        function executeRegisterVote(optionId) {
            const vd = appState.judgmentContents.voteData;
            if (vd.myVoteId !== null) {
                showToast("이미 이 사건에 대해 배심원 판결 선고를 마쳤습니다.");
                return;
            }

            vd.myVoteId = optionId;
            const targetOption = vd.options.find(o => o.id === optionId);
            if (targetOption) {
                targetOption.count += 1;
                vd.totalVotes += 1;
            }

            renderPage();
            showToast("귀중한 대배심 투표가 성공적으로 누적되었습니다!");
        }

        function forceRefreshVoting() {
            const vd = appState.judgmentContents.voteData;
            vd.options.forEach(opt => {
                opt.count += Math.floor(Math.random() * 5);
            });
            vd.totalVotes = vd.options.reduce((acc, opt) => acc + opt.count, 0);
            
            renderPage();
            showToast("실시간 배심원 집계 투표가 갱신되었습니다!");
        }

        function toggleJudgmentLike() {
            const vd = appState.judgmentContents.voteData;
            if (!vd.likesCount) vd.likesCount = 342;
            if (vd.liked) {
                vd.likesCount -= 1;
                vd.liked = false;
            } else {
                vd.likesCount += 1;
                vd.liked = true;
            }
            renderPage();
        }

        function addJudgmentOpinion() {
            const input = document.getElementById('judg-opinion-input');
            const txt = input.value.trim();
            if (!txt) {
                showCustomAlert("입력 확인", "유효한 의견을 작성한 뒤 제출해 주세요.");
                return;
            }

            const vd = appState.judgmentContents.voteData;
            vd.comments.unshift({
                author: appState.currentUser.username,
                content: txt,
                createdAt: "방금 전"
            });

            renderPage();
            showToast("내 의견이 배심원 게시판에 실시간 기재되었습니다.");
        }
    </script>

    <script>
        let modalConfirmCallback = null;
        let modalCancelCallback = null;

        function showCustomConfirm(title, message, confirmText = "확인", cancelText = "취소", onConfirm = null, onCancel = null) {
            document.getElementById('modal-title').textContent = title;
            document.getElementById('modal-message').textContent = message;
            
            const cBtn = document.getElementById('modal-confirm-btn');
            const xBtn = document.getElementById('modal-cancel-btn');
            
            cBtn.textContent = confirmText;
            xBtn.textContent = cancelText;
            xBtn.classList.remove('hidden');

            modalConfirmCallback = onConfirm;
            modalCancelCallback = onCancel;

            const modal = document.getElementById('modal-container');
            const card = document.getElementById('modal-card');
            
            modal.classList.remove('hidden');
            setTimeout(() => {
                card.classList.remove('scale-95', 'opacity-0');
                card.classList.add('scale-100', 'opacity-100');
            }, 50);
        }

        function showCustomAlert(title, message, confirmText = "확인", onConfirm = null) {
            document.getElementById('modal-title').textContent = title;
            document.getElementById('modal-message').textContent = message;
            
            const cBtn = document.getElementById('modal-confirm-btn');
            const xBtn = document.getElementById('modal-cancel-btn');
            
            cBtn.textContent = confirmText;
            xBtn.classList.add('hidden'); // Hide Cancel button for alerts

            modalConfirmCallback = onConfirm;
            modalCancelCallback = null;

            const modal = document.getElementById('modal-container');
            const card = document.getElementById('modal-card');
            
            modal.classList.remove('hidden');
            setTimeout(() => {
                card.classList.remove('scale-95', 'opacity-0');
                card.classList.add('scale-100', 'opacity-100');
            }, 50);
        }

        function closeModal() {
            const card = document.getElementById('modal-card');
            card.classList.remove('scale-100', 'opacity-100');
            card.classList.add('scale-95', 'opacity-0');
            setTimeout(() => {
                document.getElementById('modal-container').classList.add('hidden');
            }, 200);
        }

        function showToast(message) {
            const toast = document.getElementById('toast');
            toast.textContent = message;
            toast.classList.remove('opacity-0');
            toast.classList.add('opacity-100');
            
            setTimeout(() => {
                toast.classList.remove('opacity-100');
                toast.classList.add('opacity-0');
            }, 2500);
        }

        // Safe DOM Binder (Resolves early null reference errors by binding on DOMContentLoaded)
        document.addEventListener('DOMContentLoaded', () => {
            const confirmBtn = document.getElementById('modal-confirm-btn');
            const cancelBtn = document.getElementById('modal-cancel-btn');

            if (confirmBtn) {
                confirmBtn.addEventListener('click', () => {
                    closeModal();
                    if (modalConfirmCallback) modalConfirmCallback();
                });
            }

            if (cancelBtn) {
                cancelBtn.addEventListener('click', () => {
                    closeModal();
                    if (modalCancelCallback) modalCancelCallback();
                });
            }

            // Route to initial home page on successful load
            router.navigate('home');
        });
    </script>
