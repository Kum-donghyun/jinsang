        // 스토리 시작
        function startStory() {
            document.getElementById('intro-screen').classList.add('hidden');
            const gameScreen = document.getElementById('game-screen');
            gameScreen.classList.remove('hidden');
            gameScreen.classList.add('fade-in');
            document.getElementById('gauge-container').classList.remove('hidden');
            updateMentalGauge();
            loadStage(0);
        }

        // 스테이지 로딩
        function loadStage(stageIndex) {
            currentStage = stageIndex;
            const data = scenarios[stageIndex];

            // 정보 주입
            document.getElementById('stage-badge').innerText = data.stageBadge;
            document.getElementById('scenario-title').innerText = data.title;
            document.getElementById('scenario-desc').innerText = data.desc;
            document.getElementById('phone-caller').innerText = data.caller;
            document.getElementById('phone-status').innerText = data.status;

            // 채팅 스레드 리셋 및 채우기
            const chatThread = document.getElementById('chat-thread');
            chatThread.innerHTML = '';
            
            data.chatHistory.forEach(chat => {
                appendChatMessage(chat.sender, chat.msg, chat.time);
            });

            // 선택지 패널 채우기
            const optionsPanel = document.getElementById('options-panel');
            optionsPanel.innerHTML = '';

            data.options.forEach((opt, idx) => {
                const btn = document.createElement('button');
                btn.className = "w-full text-left p-4 bg-neutral-900 hover:bg-neutral-800 text-gray-200 rounded-xl border border-neutral-800 hover:border-orange-500/50 transition-all text-sm leading-relaxed duration-300 transform hover:translate-x-1";
                btn.innerHTML = `<span class="text-orange-500 font-bold mr-2">선택 ${idx + 1}.</span> ${opt.text}`;
                btn.onclick = () => selectOption(opt);
                optionsPanel.appendChild(btn);
            });
        }

        // 채팅 메시지 추가 유틸리티
        function appendChatMessage(sender, text, time) {
            const chatThread = document.getElementById('chat-thread');
            const messageDiv = document.createElement('div');
            
            if (sender === '학부모') {
                messageDiv.className = "flex flex-col items-start mb-4 fade-in";
                messageDiv.innerHTML = `
                    <div class="text-xs text-neutral-500 mb-1">${time} [학부모]</div>
                    <div class="bg-red-950/40 text-red-100 p-3.5 rounded-2xl rounded-tl-none max-w-[85%] text-sm leading-relaxed border border-red-900/30">
                        ${text}
                    </div>
                `;
            } else {
                messageDiv.className = "flex flex-col items-end mb-4 fade-in";
                messageDiv.innerHTML = `
                    <div class="text-xs text-neutral-500 mb-1">${time} [나 (교사)]</div>
                    <div class="bg-orange-800 text-white p-3.5 rounded-2xl rounded-tr-none max-w-[85%] text-sm leading-relaxed border border-orange-700">
                        ${text}
                    </div>
                `;
            }
            chatThread.appendChild(messageDiv);
            chatThread.scrollTop = chatThread.scrollHeight;
        }

        // 선택지 클릭 처리
        function selectOption(option) {
            // 1. 교사의 답변 전송 시각화
            appendChatMessage('교사', option.reply, '실시간 대응');
            
            // 2. 딜레이 후 학부모의 광폭한 피드백 전송
            setTimeout(() => {
                appendChatMessage('학부모', option.parentReaction, '즉시 반발');
                
                // 화면 진동 기믹 발동
                const visualPhone = document.getElementById('visual-phone');
                visualPhone.classList.add('shake-active');
                setTimeout(() => visualPhone.classList.remove('shake-active'), 500);

                // 멘탈 가감 및 업데이트
                mentalScore += option.mentalImpact;
                if (mentalScore < 0) mentalScore = 0;
                updateMentalGauge();

                // 선택지 차단 및 다음 진행 버튼 삽입
                const optionsPanel = document.getElementById('options-panel');
                optionsPanel.innerHTML = '';
                
                const nextBtn = document.createElement('button');
                nextBtn.className = "w-full py-4 bg-orange-700 hover:bg-orange-600 text-white font-bold rounded-xl transition-all flex items-center justify-center space-x-2";
                
                if (option.nextStage < scenarios.length) {
                    nextBtn.innerHTML = `<span>다음 고비 마주하기</span> <i class="fa-solid fa-arrow-right"></i>`;
                    nextBtn.onclick = () => loadStage(option.nextStage);
                } else {
                    nextBtn.innerHTML = `<span>정신적 한계 도달 (결과 확인)</span> <i class="fa-solid fa-skull-crossbones"></i>`;
                    nextBtn.onclick = () => showResult();
                }
                optionsPanel.appendChild(nextBtn);

            }, 1000);
        }

        // 멘탈 게이지 그래픽 업데이트
        function updateMentalGauge() {
            const bar = document.getElementById('mental-bar');
            const text = document.getElementById('mental-text');
            bar.style.width = mentalScore + '%';
            text.innerText = mentalScore + '%';

            if (mentalScore > 60) {
                bar.className = "bg-green-600 h-full transition-all duration-500";
                text.className = "text-xs font-bold text-green-500";
            } else if (mentalScore > 30) {
                bar.className = "bg-orange-500 h-full transition-all duration-500";
                text.className = "text-xs font-bold text-orange-500";
            } else {
                bar.className = "bg-red-600 h-full transition-all duration-500 animate-pulse";
                text.className = "text-xs font-bold text-red-500 animate-bounce";
            }
        }

        // 결과 페이지 노출
        function showResult() {
            document.getElementById('game-screen').classList.add('hidden');
            document.getElementById('gauge-container').classList.add('hidden');
            
            const resultScreen = document.getElementById('result-screen');
            resultScreen.classList.remove('hidden');
            resultScreen.classList.add('fade-in');

            const endingTitle = document.getElementById('ending-title');
            const endingDesc = document.getElementById('ending-desc');

            // 멘탈 스코어에 따른 멀티 엔딩 멘트 조율
            if (mentalScore > 40) {
                endingTitle.innerText = "상처투성이의 간신히 버틴 교탁";
                endingDesc.innerText = "선생님은 어찌어찌 버텨냈지만 이미 심장은 깊은 내상을 입었습니다. 밤마다 걸려오는 기습 민원 전화 소리가 환청처럼 귓전을 맴돕니다. 언제 무너질지 모르는 외줄 타기가 매일 반복됩니다.";
            } else {
                endingTitle.innerText = "결국 바닥에 부러진 분필";
                endingDesc.innerText = "아무런 보호 장치가 없는 교실에서 당신은 완전히 연소해 버렸습니다. 학부모의 일방적인 분풀이, 누구도 구해주지 않는 밀폐된 공간에서 교사라는 꿈은 악몽으로 뒤바뀐 채 마침표를 찍었습니다.";
            }
        }

        // 리스타트
        function restartStory() {
            mentalScore = 100;
            currentStage = 0;
            document.getElementById('result-screen').classList.add('hidden');
            startStory();
        }
